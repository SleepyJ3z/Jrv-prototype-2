const express = require('express');
const nodemailer = require('nodemailer');
const app = express();

app.use(express.json());

// Gmail SMTP Transporter Configuration
const transporter = nodemailer.createTransport({
  service: 'gmail',
  auth: {
    user: process.env.GMAIL_USER, // jrvofficefurnituresandservices@gmail.com
    pass: process.env.GMAIL_APP_PASSWORD
  }
});

app.post('/api/inquire', async (req, res) => {
  const { name, email, phone, service, location, message } = req.body;

  const mailOptions = {
    from: `"${name}" <${email}>`,
    to: 'jrvofficefurnituresandservices@gmail.com',
    subject: `New Fit-Out Inquiry: ${service}`,
    html: `
      <div style="font-family: Arial, sans-serif; padding: 20px;">
        <h2 style="color: #274E39;">JRV's Office Inquiry</h2>
        <p><strong>Client Name:</strong> ${name}</p>
        <p><strong>Email:</strong> ${email}</p>
        <p><strong>Phone/Viber:</strong> ${phone}</p>
        <p><strong>Service Required:</strong> ${service}</p>
        <p><strong>Location:</strong> ${location}</p>
        <hr />
        <p><strong>Message Details:</strong></p>
        <p>${message}</p>
      </div>
    `
  };

  try {
    await transporter.sendMail(mailOptions);
    res.status(200).json({ success: true, message: 'Email delivered to JRV inbox!' });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});
